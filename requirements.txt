# =============================================================================
# 适用于：Sphinx + Markdown + Breathe + Read the Docs 的 SoC 文档工程
# 输出格式：HTML / PDF / ePub
# 使用方式：pip install -r requirements.txt
# =============================================================================

# ----1. 核心文档生成引擎------------------------------------------------------
sphinx>=7.2,<8.0

# ----2. Read the Docs 同款主题（与 Zephyr 一致）-------------------------------
# 需要与conf.py中的html_theme对应的字符串名称保持一致
sphinx-rtd-theme>=1.3.0
# sphinx-book-theme 和sphinx-rtd-theme二选一即可
sphinx-book-theme>=1.0.0

# ----3. Markdown 解析支持（MyST 解析器，替代 reStructuredText）---------------
# 需要在conf.py的extension中配置：'myst_parser'
# --source_suffix = {'.rst': 'restructuredtext','.md': 'markdown',}
myst-parser>=2.0.0

# ----4. 从源码注释生成 API 文档（连接 Doxygen XML）-----------------------------
# 在conf.py文件的extension中新增：'breathe'
# --breathe_projects = {"MyProject": "./doxygen_output/xml/"}
# --breathe_default_project = "MyProject"
# --breathe_default_members = ('members', 'undoc-members')
# 同时，在.readthedocs.yaml 的 apt_packages 中安装 doxygen
breathe>=4.35.0

# ----5. （可选）增强文档体验的扩展-----------------------------------------------
# 标签页组件，用于多语言代码示例、不同版本对比等
# 在conf.py中需要在extension中配置：'sphinx_tabs.tabs'
sphinx-tabs>=3.4.0

# ----6. 代码块一键复制按钮-------------------------------------------------------
# 在conf.py中需要在extension中配置'sphinx_copybutton'
sphinx-copybutton>=0.5.0

# ----7. Mermaid 流程图/时序图支持------------------------------------------------
# 需要在conf.py的extension中配置'sphinxcontrib.mermaid'
sphinxcontrib-mermaid>=0.9.0

# ----8. PlantUML 图表支持（直接在 Markdown/reST 中嵌入 plantuml 代码块）---------
# 需要在conf.py的extension中新增：'sphinxcontrib.plantuml'
# --plantuml_output_format = 'svg'
# 同时在.readthedocs.yaml 的 apt_packages 中安装 plantuml 和 graphviz
sphinxcontrib-plantuml>=0.26

# ----9. PDF 和 ePub 生成注意事项：----------------------------------------------
# PDF 依赖系统 LaTeX 环境（在 .readthedocs.yaml 中通过 apt_packages 安装），
# 不需要额外的 Python 包。
# ePub 由 Sphinx 内置构建器直接生成，同样无需额外 Python 依赖。
# 如果你的文档含 SVG 图片，建议添加：
# 在conf.py的extension中新增：'sphinxcontrib.svg2pdfconverter'
# --svg2pdf_converter = 'cairosvg'
# sphinxcontrib-svg2pdfconverter>=1.2.0

# ----10. 文档中外部链接检查（避免死链）------------------------------------------
# conf.py中可配置：sphinx_tabs_valid_builders = ['linkcheck']
sphinx-ext-linkcheck>=1.0.0

# ----11. 制作多语言版本文档------------------------------------------------------
# 提供命令行工具（如 sphinx-intl update 和 sphinx-intl build），用于提取原文中的待翻译文本（生成 .pot 文件）
# 需要在conf.py中新增：
# --language = 'zh_CN'  # 或 'en'
# --locale_dirs = ['locales/']
# --gettext_compact = False
sphinx-intl

# ----12. 自动将纯文本 URL 转换为可点击的超链接-----------------------------------
# 需要在conf.py的extension中配置：'linkify'
linkify-it-py

# ----13. 自动生成 API 变更日志（如果使用版本管理）--------------------------------
# 需要在conf.py的extension中新增：'sphinx_changelog'
#sphinx-changelog>=1.4.0
